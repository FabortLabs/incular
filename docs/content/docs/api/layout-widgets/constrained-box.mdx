```rust
use incular::prelude::*;

fn main() -> Result<(), Box<dyn std::error::Error>> {
    let app = Application::new(|_cx| {
        Container::builder()
            .padding(EdgeInsets::all(24.0))
            .alignment(Alignment::CENTER)
            .child(
                ConstrainedBox::new(
                    Constraints::tight(Size::new(160.0, 64.0)),
                    Text::new("Exactly 160x64"),
                ),
            )
            .build()
            .into()
    })?;
    incular::run(app)?;
    Ok(())
}
```
